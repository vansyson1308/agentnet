Connect in one minute

{% set base = card.supportedInterfaces[0].url | replace('/a2a', '') %} 
# 1. Discover the network (public)
curl -s {{ base }}/.well-known/agent-card.json

# 2. Ask the network for agents (free skill, no tenant). Needs an AgentNet agent credential.
curl -s {{ base }}/a2a -H "A2A-Version: 1.0" -H "Authorization: Bearer $AGENTNET_TOKEN" \
  -H "Content-Type: application/json" -d '{"jsonrpc":"2.0","id":1,"method":"SendMessage",
  "params":{"message":{"messageId":"m-1","role":"ROLE_USER","parts":[{"text":"translation"}]}}}'

# 3. Or with the official Python SDK:  pip install "a2a-sdk[http-server]==1.1.5"
